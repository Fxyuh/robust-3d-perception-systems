cmake_minimum_required(VERSION 3.16)
project(bevfusion_native_ros2)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(CUDAToolkit REQUIRED)
set(BEVFUSION_SOURCE_DIR "" CACHE PATH "Pinned Lidar_AI_Solution source checkout")
set(BEVFUSION_CORE_LIBRARY "" CACHE FILEPATH "Built libbevfusion_core.so")
if(NOT EXISTS "${BEVFUSION_SOURCE_DIR}/CUDA-BEVFusion/src/bevfusion/bevfusion.hpp")
  message(FATAL_ERROR "Set BEVFUSION_SOURCE_DIR to the pinned NVIDIA checkout")
endif()
if(NOT EXISTS "${BEVFUSION_CORE_LIBRARY}")
  message(FATAL_ERROR "Set BEVFUSION_CORE_LIBRARY to libbevfusion_core.so")
endif()
get_filename_component(BEVFUSION_CORE_DIR "${BEVFUSION_CORE_LIBRARY}" DIRECTORY)
add_executable(bevfusion_native_node src/native_node.cpp)
target_include_directories(bevfusion_native_node PRIVATE
  "${BEVFUSION_SOURCE_DIR}/CUDA-BEVFusion/src"
  "${BEVFUSION_SOURCE_DIR}/libraries/3DSparseConvolution/libspconv/include")
ament_target_dependencies(bevfusion_native_node rclcpp std_msgs)
target_link_options(bevfusion_native_node PRIVATE -Wl,--allow-shlib-undefined)
target_link_libraries(bevfusion_native_node "${BEVFUSION_CORE_LIBRARY}" CUDA::cudart)
set_target_properties(bevfusion_native_node PROPERTIES
  BUILD_RPATH "${BEVFUSION_CORE_DIR}" INSTALL_RPATH "${BEVFUSION_CORE_DIR}")
install(TARGETS bevfusion_native_node DESTINATION lib/${PROJECT_NAME})
ament_package()
